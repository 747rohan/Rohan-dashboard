===== dash-phase-shim.service
[Unit]
Description=Dashboard phase shim (7RL metrics.jsonl -> orchestrator.db + phases ingest)
After=network-online.target docker.service

[Service]
Type=simple
User=ubuntu
EnvironmentFile=/opt/dashboard/dashboard.env
ExecStart=/usr/bin/python3 /opt/dashboard/shim/phase_shim.py
Restart=always
RestartSec=10
Nice=10

[Install]
WantedBy=multi-user.target
